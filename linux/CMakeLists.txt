cmake_minimum_required(VERSION 3.22)
project(TuxSSTV VERSION 1.15.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)

find_package(Qt6 6.2 REQUIRED COMPONENTS Core Gui Widgets Concurrent Network)
find_library(HAMLIB_LIBRARY NAMES hamlib libhamlib.so.4 REQUIRED)

target_link_libraries(radiocore PUBLIC Qt6::Core ${HAMLIB_LIBRARY})
target_include_directories(radiocore PUBLIC src)
target_compile_options(radiocore PRIVATE -Wall -Wextra -Wpedantic)
add_library(sstvcore src/core/spectrum.cpp src/core/spectrum.h src/core/audio_levels.cpp src/core/audio_levels.h src/core/modes.cpp src/core/patterns.cpp src/core/codec.cpp src/core/fskid.cpp src/core/receiver.cpp src/core/morse.cpp src/core/composition.cpp src/core/templates.cpp src/core/layouts.cpp src/core/exchangetemplates.cpp src/core/wav.cpp)

target_sources(sstvcore PRIVATE src/emojis.cpp src/emojis.h resources/emoji_catalog.qrc)
target_include_directories(sstvcore PUBLIC src)
target_sources(sstvcore PRIVATE src/core/televisiontemplates.cpp src/core/televisiontemplates.h)
target_link_libraries(sstvcore PUBLIC Qt6::Core Qt6::Gui)
target_compile_options(sstvcore PRIVATE -Wall -Wextra -Wpedantic)
add_library(sstvui src/mainwindow_audio.cpp src/mainwindow_edits.cpp src/tx_edit_state.h src/audio_levels_panel.cpp src/audio_levels_panel.h src/radio_frontend.cpp src/radio_frontend.h src/insetdialog.cpp src/insetdialog.h src/patternsdialog.cpp src/patternsdialog.h src/contactlog.cpp src/contactlog.h src/contactlog_ui.cpp src/contactlog_ui.h src/audio_wizard.cpp src/audio_wizard.h src/playback_mixer.cpp src/playback_mixer.h src/mainwindow.cpp src/mainwindow_ui.cpp src/mainwindow_radio.cpp src/mainwindow.h src/radio_ui.cpp src/radio_ui.h src/composer.cpp src/composer.h src/captionpreview.cpp src/captionpreview.h src/templatesdialog.cpp src/templatesdialog.h src/skins.cpp src/skins.h src/imageview.cpp src/imageview.h src/signalview.cpp src/signalview.h resources/skin_icons.qrc)
target_sources(sstvui PRIVATE src/mainwindow_profiles.cpp)
target_link_libraries(sstvui PUBLIC sstvcore radiocore Qt6::Widgets Qt6::Concurrent)
target_compile_options(sstvui PRIVATE -Wall -Wextra -Wpedantic)
target_sources(sstvui PRIVATE src/selfie_dialog.cpp src/selfie_dialog.h)
target_sources(sstvui PRIVATE src/remote_station.cpp src/remote_station.h src/mainwindow_remote.cpp)
target_sources(sstvui PRIVATE src/transmit_monitor.cpp src/transmit_monitor.h)
target_sources(sstvui PRIVATE src/voice_audio.cpp src/voice_audio.h src/mainwindow_voice.cpp)
target_sources(sstvui PRIVATE src/mic_check_dialog.cpp src/mic_check_dialog.h)
target_sources(sstvui PRIVATE src/mainwindow_radio_setup.cpp)
target_sources(sstvui PRIVATE src/audio_wizard_relay.cpp src/mainwindow_audio_setup.cpp)
target_sources(sstvui PRIVATE src/mainwindow_setup_access.cpp)
target_sources(sstvui PRIVATE src/mainwindow_setup_wizard.cpp)
target_sources(sstvui PRIVATE src/branding.cpp src/branding.h)
target_sources(sstvui PRIVATE resources/tux_pictures.qrc)
target_link_libraries(sstvui PUBLIC Qt6::Network)
add_executable(tux-sstv src/main.cpp)
add_custom_command(TARGET tux-sstv POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E rm -f "$<TARGET_FILE_DIR:tux-sstv>/opensstv"
  COMMAND ${CMAKE_COMMAND} -E create_symlink "$<TARGET_FILE_NAME:tux-sstv>" "$<TARGET_FILE_DIR:tux-sstv>/opensstv")
add_custom_target(opensstv DEPENDS tux-sstv)
target_link_libraries(tux-sstv PRIVATE sstvui)
target_compile_definitions(tux-sstv PRIVATE TUXSSTV_VERSION="${PROJECT_VERSION}")
target_compile_options(tux-sstv PRIVATE -Wall -Wextra -Wpedantic)
